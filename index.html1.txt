  <script>
    // Create background floating hearts
    const heartBg = document.getElementById('heartBg');
    const heartIcons = ['💖', '💕', '🌸', '✨', '💗'];
    for(let i=0; i<15; i++) {
      const heart = document.createElement('div');
      heart.className = 'bg-heart';
      heart.innerText = heartIcons[Math.floor(Math.random() * heartIcons.length)];
      heart.style.left = Math.random() * 100 + 'vw';
      heart.style.animationDelay = (Math.random() * 8) + 's';
      heart.style.fontSize = (Math.random() * 20 + 16) + 'px';
      heartBg.appendChild(heart);
    }

    // Runaway Evasive Button Logic with Micro Toasts
    const runawayBtn = document.getElementById('runawayBtn');
    const toast = document.getElementById('toast');
    const toastMsg = document.getElementById('toastMsg');
    
    const teaseMessages = [
      "Nice try! 😜",
      "You can't escape my love! 🥰",
      "Nope! Try the red button! 💕",
      "Wrong button, sweetie! 💖",
      "Only forgiveness allowed here! 🌸"
    ];

    function moveButton() {
      const x = (Math.random() - 0.5) * 220;
      const y = (Math.random() - 0.5) * 160;
      runawayBtn.style.transform = `translate(${x}px, ${y}px)`;
      
      // Show cute playful toast
      const randomMsg = teaseMessages[Math.floor(Math.random() * teaseMessages.length)];
      toastMsg.innerText = randomMsg;
      toast.classList.add('show');
      
      setTimeout(() => {
        toast.classList.remove('show');
      }, 1800);
    }

    // Polaroid card flip
    function flipCard(card) {
      const title = card.children[1];
      const note = card.children[2];
      if(note.classList.contains('hidden')) {
        note.classList.remove('hidden');
        title.classList.add('hidden');
      } else {
        note.classList.add('hidden');
        title.classList.remove('hidden');
      }
    }

    // Forgive Action & Confetti
    function forgiveAction() {
      confetti({
        particleCount: 120,
        spread: 80,
        origin: { y: 0.6 }
      });
      document.getElementById('forgiveModal').classList.remove('hidden');
    }

    function closeModal() {
      document.getElementById('forgiveModal').classList.add('hidden');
    }
